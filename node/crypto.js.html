1 import { log } from 'node:console';
 2 import crypto from 'node:crypto';
 3 
 4 // 对称加密算法
 5 
 6 let key = crypto.randomBytes(32);
 7 
 8 let iv = Buffer.from(crypto.randomBytes(16));
 9 
10 // 1 algorithm 接收加密算法
11 // 2 key 密钥 32位
12 // 3 iv 初始化向量 16位 保证生成的密钥串不一样 也可进行补位
13 const cipher = crypto.createCipheriv("aes-256-cbc",key,iv);
14 
15 // hex 16进制
16 cipher.update("some thing","utf-8","hex")
17 
18 const result = cipher.final("hex")
19 
20 console.log("result",result)
21 
22 // 解密
23 
24 const deCipher = crypto.createDecipheriv("aes-256-cbc",key,iv);
25 
26 deCipher.update(result,"hex","utf-8");
27 
28 console.log("result",deCipher.final("utf-8"));
29 
30 // 非对称加密算法
31 // 生成公钥和私钥
32 // 私钥只能由管理员拥有
33 // 公钥可以对外公开
34 
35 const {privateKey,publicKey} = crypto.generateKeyPairSync("rsa",{
36         modulusLength:2048, // 长度越长越安全，越慢
37 });
38 
39 const encrypted = crypto.publicEncrypt(publicKey,Buffer.from("hello some thing"));
40 
41 console.log("encrypted\n",encrypted.toString("hex"))
42 
43 const decrypted = crypto.privateDecrypt(privateKey,encrypted);
44 
45 console.log("decrypted\n",decrypted.toString())
46 
47 // 哈希函数
48 // 不能被解密 单向 不可逆
49 // 具有唯一性
50 // 不能明文存储
51 // 校验文件一致性
52 let hash = crypto.createHash('sha56');
53 
54 hash.update("hello hash");
55 
56 console.log("hash result" ,hash.digest("hex"));
